  <script>
    let currentRoom = "1234";
    let isBurmese = true;
    let audioCtx = null;
    let mediaRecorder = null;
    let audioChunks = [];
    let isRecording = false;

    // Web Audio Sound Generator
    function playBeep(freq = 600, duration = 0.1) {
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.frequency.value = freq;
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        gain.gain.exponentialRampToValueAtTime(0.00001, audioCtx.currentTime + duration);
        osc.stop(audioCtx.currentTime + duration);
      } catch(e) {}
    }

    function toggleLang() {
      isBurmese = !isBurmese;
      document.getElementById('msgInput').placeholder = isBurmese ? "စာရိုက်ပါ..." : "Type a message...";
      document.getElementById('roomStatus').innerText = (isBurmese ? "အခန်း: " : "Room: ") + currentRoom;
    }

    function joinRoom() {
      const val = document.getElementById('roomInput').value.trim();
      if (val) currentRoom = val;
      document.getElementById('roomStatus').innerText = (isBurmese ? "အခန်း: " : "Room: ") + currentRoom;
      document.getElementById('roomModal').style.display = 'none';
      setupBroadcast();
    }

    let channel = null;
    function setupBroadcast() {
      channel = new BroadcastChannel("pwa_chat_" + currentRoom);
      channel.onmessage = (e) => {
        appendMessage(e.data, 'other');
        playBeep(800, 0.15);
      };
    }

    function appendMessage(data, sender) {
      const box = document.getElementById('chatBox');
      const div = document.createElement('div');
      div.className = 'msg ' + sender;

      if (data.type === 'text') {
        div.innerText = data.content;
      } else if (data.type === 'image') {
        const img = document.createElement('img');
        img.src = data.content;
        div.appendChild(img);
      } else if (data.type === 'audio') {
        const audio = document.createElement('audio');
        audio.src = data.content;
        audio.controls = true;
        div.appendChild(audio);
      }
      box.appendChild(div);
      box.scrollTop = box.scrollHeight;
    }

    function sendText() {
      const input = document.getElementById('msgInput');
      const text = input.value.trim();
      if (!text) return;
      const data = { type: 'text', content: text };
      if (channel) channel.postMessage(data);
      appendMessage(data, 'me');
      playBeep(400, 0.08);
      input.value = "";
    }

    function handleKey(e) {
      if (e.key === 'Enter') sendText();
    }

    function sendImage(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(evt) {
        const data = { type: 'image', content: evt.target.result };
        if (channel) channel.postMessage(data);
        appendMessage(data, 'me');
      };
      reader.readAsDataURL(file);
    }

    async function toggleRecord() {
      const recBtn = document.getElementById('recBtn');
      if (!isRecording) {
        try {
          const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
          mediaRecorder = new MediaRecorder(stream);
          audioChunks = [];
          mediaRecorder.ondataavailable = e => audioChunks.push(e.data);
          mediaRecorder.onstop = () => {
            const blob = new Blob(audioChunks, { type: 'audio/webm' });
            const reader = new FileReader();
            reader.onload = function(evt) {
              const data = { type: 'audio', content: evt.target.result };
              if (channel) channel.postMessage(data);
              appendMessage(data, 'me');
            };
            reader.readAsDataURL(blob);
          };
          mediaRecorder.start();
          isRecording = true;
          recBtn.classList.add('recording');
        } catch(err) {
          alert(isBurmese ? "မိုက်ခရိုဖုန်း ခွင့်ပြုချက် လိုအပ်ပါသည်။" : "Microphone permission required.");
        }
      } else {
        mediaRecorder.stop();
        isRecording = false;
        recBtn.classList.remove('recording');
      }
    }

    function startCall() {
      alert(isBurmese ? "Voice Call စနစ်ကို စမ်းသပ်နေပါသည်..." : "Initiating Voice Call interface...");
    }

    // PWA Service Worker Registration
    if ('serviceWorker' in navigator) {
      const swCode = "self.addEventListener('install', e => e.waitUntil(self.skipWaiting()));";
      const blob = new Blob([swCode], { type: 'application/javascript' });
      navigator.serviceWorker.register(URL.createObjectURL(blob)).catch(()=>{});
    }
  </script>
